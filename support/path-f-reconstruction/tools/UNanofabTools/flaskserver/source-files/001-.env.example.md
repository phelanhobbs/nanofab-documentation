```text
# Flask Configuration
FLASK_APP=run.py
# Production is the secure default (Duo 2FA on, Secure cookies). Set this to
# `development` ONLY for local testing.
FLASK_ENV=production
# REQUIRED in production — the app refuses to start with an unset/default value.
# Generate one with: python -c "import secrets; print(secrets.token_hex(32))"
SECRET_KEY=your-secret-key-here-change-in-production

# Debug Mode (NEVER set to True in production)
DEBUG_MODE=False

# Server Configuration
# Flask binds to loopback; nginx terminates TLS on 443 and proxies here.
# Do NOT bind Flask directly to the external IP / 443.
HOST=127.0.0.1
PORT=5000

# SSL Certificate Paths (only used if running Flask standalone without nginx)
SSL_CERT_PATH=/path/to/cert.pem
SSL_KEY_PATH=/path/to/key.pem

# Database Configuration (SQLite app databases)
DATABASE_URI=sqlite:///signininfo.db
SESSION_DATABASE_URI=sqlite:///sessioninfo.db
TASK_DATABASE_URI=sqlite:///tasks.db
PARTICLE_SENSOR_DATABASE_URI=sqlite:///particle_sensors.db

# Chemical Inventory (PostgreSQL)
CHEM_PGHOST=localhost
CHEM_PGPORT=5432
CHEM_POSTGRES_DB=cheminventory
CHEM_POSTGRES_USER=postgres
CHEM_POSTGRES_PASSWORD=<redacted-secret-value>
# Shared HMAC secret for the WordPress -> chem-inventory signed link gate.
# Must match the value in the WordPress 'chem_inventory_link' snippet.
CHEM_SSO_SECRET=<redacted-secret-value>

# Duo Security Configuration (real values from the Duo Admin Panel -> Auth API)
DUO_IKEY=your-duo-integration-key
DUO_SKEY=your-duo-secret-key
DUO_HOST=your-duo-api-host

# Session Configuration
SESSION_COOKIE_SECURE=True
SESSION_COOKIE_HTTPONLY=True
SESSION_COOKIE_SAMESITE=Lax
PERMANENT_SESSION_LIFETIME=7200

# File Upload Configuration
UPLOAD_FOLDER=uploads
MAX_CONTENT_LENGTH=16777216
ALLOWED_EXTENSIONS=txt,pdf,csv,png,jpg,jpeg,doc,docx,xls,xlsx

# Data Directory
DATA_DIR=HSCDATA
LOG_DATA_DIR=LogData

# CORES API bearer <redacted-bearer-token> (raw token; used by HSCDownloader / PreciousMetalReader)
CORES_TOKEN=<redacted-secret-value>
```
